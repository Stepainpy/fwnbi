template <class CharT, class Traits, size_t B, class D, bool S>
basic_istream<CharT, Traits>& operator>>(
    basic_istream<CharT, Traits>& is, fwnbi::basic_integer<B, D, S>& value
);

Formatted input from std::basic_istream.

Example 
#include "fwnbi.hpp"
#include <iostream>
#include <sstream>

int main() {
    std::istringstream iss("   0000012345jrl");
    fwnbi::uint128_t value;
    iss >> std::hex >> value;
    std::cout << "Received value: " << value << '\n';
}
Output 
Received value: 74565
